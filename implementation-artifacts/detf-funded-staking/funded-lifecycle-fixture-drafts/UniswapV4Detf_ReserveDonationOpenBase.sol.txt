/// @dev Holds an open PoolManager unlock and calls NFT donate from unlockCallback (DN22).
contract Uv4DetfDonateDuringUnlockHarness is IUnlockCallback {
    IPoolManager public immutable pm;

    constructor(IPoolManager pm_) {
        pm = pm_;
    }

    function run(address target, bytes calldata data) external returns (bytes memory) {
        return pm.unlock(abi.encode(target, data));
    }

    function unlockCallback(bytes calldata raw) external returns (bytes memory) {
        require(msg.sender == address(pm), "not pm");
        (address target, bytes memory data) = abi.decode(raw, (address, bytes));
        (bool ok, bytes memory ret) = target.call(data);
        if (!ok) {
            assembly {
                revert(add(ret, 32), mload(ret))
            }
        }
        return ret;
    }
}

/// @dev Independent custody assertions shared by native-unit and provider donation fixtures.
abstract contract V4FundedDonationAssertions is Test {
    struct DnLiveSnap {
        uint256 principal;
        uint256 positionGons;
        uint256 supply;
        uint256 protocolLp;
        uint256 userDetf;
        uint256 backing;
        bytes32 stakingState;
    }

    function _fundedDonationSnapshot(IUniswapV4Detf info_, uint256 id_, address user_)
        internal view returns (DnLiveSnap memory s_)
    {
        IDetfBondNFT nft_ = IDetfBondNFT(info_.bondNftVault());
        IStakedDETF staking_ = IStakedDETF(info_.rebasingClaimToken());
        DETFFundedStakingMath.BondPosition memory p_ = nft_.positionOf(id_);
        s_.principal = p_.principal;
        s_.positionGons = p_.stakingGons;
        s_.supply = IERC20(address(info_)).totalSupply();
        s_.protocolLp = IERC20(info_.hook()).balanceOf(address(nft_)) + IERC20(info_.hook()).balanceOf(address(info_));
        s_.userDetf = IERC20(address(info_)).balanceOf(user_);
        s_.backing = IERC20(address(info_)).balanceOf(address(staking_));
        s_.stakingState = keccak256(abi.encode(staking_.stakingState()));
    }

    function _assertFundedDonationOnly(IUniswapV4Detf info_, uint256 id_, address user_, DnLiveSnap memory before_, uint256 lp_)
        internal view
    {
        DnLiveSnap memory after_ = _fundedDonationSnapshot(info_, id_, user_);
        assertGt(lp_, 0, "donation returns acquired LP");
        assertGe(after_.protocolLp, before_.protocolLp + lp_, "actual protocol custody includes donation and any swept LP");
        assertEq(after_.principal, before_.principal, "reserve gift cannot change purchased principal");
        assertEq(after_.positionGons, before_.positionGons, "reserve gift cannot mint staking ownership");
        assertEq(after_.supply, before_.supply, "reserve donation issues no DETF");
        assertEq(after_.backing, before_.backing, "reserve gift is separate from funded staking backing");
        assertEq(after_.stakingState, before_.stakingState, "donation preserves funded ledger and standing rights");
    }

    function _claimFundedDonationPosition(IUniswapV4Detf info_, uint256 id_, address holder_) internal returns (uint256 paid_) {
        IDETFFundedRewards(address(info_)).synchronizeRewards();
        IDetfBondNFT nft_ = IDetfBondNFT(info_.bondNftVault());
        IStakedDETF receipt_ = IStakedDETF(info_.rebasingClaimToken());
        DETFFundedStakingMath.BondClaim memory quote_ = nft_.previewClaim(id_);
        uint256 lpBefore_ = IERC20(info_.hook()).balanceOf(address(nft_)) + IERC20(info_.hook()).balanceOf(address(info_));
        uint256 before_ = receipt_.balanceOf(holder_);
        vm.prank(holder_);
        (uint256 principal_, uint256 rewards_) = nft_.claimBond(id_, holder_);
        assertEq(principal_, quote_.principalDue, "donated reserve cannot alter principal preview");
        assertEq(rewards_, quote_.rewardsDue, "funded reward preview equals execution");
        paid_ = principal_ + rewards_;
        assertGt(paid_, 0, "funded payout");
        assertEq(receipt_.balanceOf(holder_) - before_, paid_, "only funded sDETF paid");
        assertEq(IERC20(info_.hook()).balanceOf(address(nft_)) + IERC20(info_.hook()).balanceOf(address(info_)), lpBefore_, "bond claim retains entire protocol LP");
    }

    function _donationUnstake(IUniswapV4Detf info_, address holder_) internal returns (uint256 amount_) {
        IStakedDETF receipt_ = IStakedDETF(info_.rebasingClaimToken());
        amount_ = receipt_.balanceOf(holder_);
        uint256 rawBefore_ = IERC20(address(info_)).balanceOf(holder_);
        vm.prank(holder_);
        uint256 out_ = receipt_.exchangeIn(IERC20(address(receipt_)), amount_, IERC20(address(info_)), amount_, holder_, false, block.timestamp);
        assertEq(out_, amount_, "exact held-DETF unstake");
        assertEq(IERC20(address(info_)).balanceOf(holder_) - rawBefore_, amount_, "funded raw payout");
    }

    function _assertDonationStanding(IUniswapV4Detf info_, address oracle_) internal view {
        IStakedDETF receipt_ = IStakedDETF(info_.rebasingClaimToken());
        IStakedDETF.StakingState memory s_ = receipt_.stakingState();
        (, uint256 f_, uint256 c_) = IVaultFeeOracleQuery(oracle_).seigniorageSplitOfVault(address(info_));
        uint256 target_ = Math.mulDiv(s_.totalGons, 1e18, 1e18 - f_ - c_);
        assertGe(s_.feeWeight, Math.mulDiv(target_, f_, 1e18), "standing fee weight is topped up and never reduced");
        assertGe(s_.creatorWeight, Math.mulDiv(target_, c_, 1e18), "standing creator weight is topped up and never reduced");
        assertGe(s_.accountedBacking, receipt_.totalSupply(), "actual funded liability");
        assertGe(IERC20(address(info_)).balanceOf(address(receipt_)), s_.accountedBacking, "actual raw backing covers accounted funding");
    }
}

    function _fundOpenPair(address to_, uint256 amount_) internal virtual {
        IERC20 tok_ = _openPairToken();
        try SimpleMintableERC20(address(tok_)).mint(to_, amount_) {
            return;
        } catch {}
        uint256 have_ = tok_.balanceOf(to_);
        if (have_ >= amount_) return;
        deal(address(tok_), to_, have_ + amount_);

    function _ensureDonor() internal {
        if (dnDonor != address(0)) return;
        dnDonorPk = 0xA11CE;
        dnDonor = vm.addr(dnDonorPk);
        IERC20 tok_ = _openPairToken();
        _fundOpenPair(dnDonor, 10_000 ether);
        address hook_ = detfInfo.hook();
        if (hook_ != address(0)) {
            address[] memory toks_ = IUniswapV4SeBufferHook(hook_).tokens();

                if (toks_[i] == detf) continue;
                try SimpleMintableERC20(toks_[i]).mint(dnDonor, 10_000 ether) {} catch {
                    uint256 have_ = IERC20(toks_[i]).balanceOf(dnDonor);
                    deal(toks_[i], dnDonor, have_ + 10_000 ether);
                }
                vm.prank(dnDonor);
                IERC20(toks_[i]).approve(address(_nft()), type(uint256).max);
                vm.prank(dnDonor);
                IERC20(toks_[i]).approve(detf, type(uint256).max);
            }
        }
        vm.prank(dnDonor);
        tok_.approve(address(_nft()), type(uint256).max);
        vm.prank(dnDonor);
        tok_.approve(detf, type(uint256).max);
    }

    function _signPermit2(
        uint256 pk_,
        address token_,
        uint256 amount_,
        address spender_,
        uint256 nonce_,
        uint256 deadline_
    ) internal view returns (bytes memory sig_) {
        bytes32 tokenHash_ = keccak256(abi.encode(TOKEN_PERMISSIONS_TYPEHASH, token_, amount_));
        bytes32 structHash_ =
            keccak256(abi.encode(PERMIT_TRANSFER_FROM_TYPEHASH, tokenHash_, spender_, nonce_, deadline_));
        bytes32 digest_ = keccak256(abi.encodePacked("\x19\x01", permit2.DOMAIN_SEPARATOR(), structHash_));
        (uint8 v_, bytes32 r_, bytes32 s_) = vm.sign(pk_, digest_);
        sig_ = abi.encodePacked(r_, s_, v_);
    }
