    function test_balancer_buffered_dust_cannot_fund_pretransfer() public {
        IBalancerHook h = _rehearseBalancer(2, 3);
        address input = h.token(1);
        address output = h.token(0);
        SimpleMintableERC20(input).mint(address(h), 10);
        _balancerSwapRoute(h, output, input);
        assertEq(IERC20(input).balanceOf(address(h)), 10, "retained reserve dust");
        assertGt(h.previewSwapExactIn(input, output, 10), 0, "dust could otherwise buy output");
        uint256 beforeBalance = IERC20(output).balanceOf(address(this));
        vm.expectRevert(abi.encodeWithSelector(ISecurePullErrors.TransferDeltaInsufficient.selector, 10, 0));
        IStandardExchangeIn(address(h)).exchangeIn(
            IERC20(input), 10, IERC20(output), 0, address(this), true, block.timestamp + 1
        );
        uint256 required = h.previewSwapExactOut(input, output, 1);
        assertLe(required, 10, "retained balance covers exact-out quote");
        vm.expectRevert(abi.encodeWithSelector(ISecurePullErrors.TransferDeltaInsufficient.selector, required, 0));
        IStandardExchangeOut(address(h)).exchangeOut(
            IERC20(input), required, IERC20(output), 1, address(this), true, block.timestamp + 1
        );
        assertEq(IERC20(output).balanceOf(address(this)), beforeBalance);
    }

    function _rehearseBalancer(uint256 n, uint256 backend, bool mixed) internal returns (IBalancerHook h) {
        IBalancerPkg.PkgArgs memory args;
        args.poolManager = address(pm);
        args.feeOracle = address(indexedexManager);
        args.baseAmp = 100;
        args.tokens = new address[](n);
        args.standardExchanges = new address[](n);
        args.rateProviders = new address[](n);
        for (uint256 i; i < n; ++i) args.tokens[i] = address(new SimpleMintableERC20("Stable asset", "STABLE"));
        for (uint256 i; i < n; ++i) {
            for (uint256 j = i + 1; j < n; ++j) {
                if (args.tokens[j] < args.tokens[i]) (args.tokens[i], args.tokens[j]) = (args.tokens[j], args.tokens[i]);
            }
            if (!mixed || i == n - 1) {
                args.standardExchanges[i] = _deploySe(backend, args.tokens[i], address(new SimpleMintableERC20("Counter asset", "OTHER")));
            }
        }
        args.tokenDecimals = Decimals.tokenDecimals(args.tokens);
        args.seDecimals = Decimals.seDecimals(args.standardExchanges);
        IBalancerPkg pkg = IBalancerPkg(_pin("phase06_stage09_balancer_stable_hook_pkg.json", ".balancerStableHookPkg", true));
        uint256 nonce = BalancerFactory.findMineNonce(hookFactory, pkg, args);
        address instance = pkg.deployVault{gas: TX_GAS}(args, nonce);
        assertEq(uint160(instance) & ((1 << 14) - 1), BalancerFactory.requiredFlags(), "hook flags");
        IUniswapV4HookStagedPairInit staged = IUniswapV4HookStagedPairInit(instance);
        for (uint256 i; i < n; ++i) {
            for (uint256 j = i + 1; j < n; ++j) staged.deployPair{gas: TX_GAS}(args.tokens[i], args.tokens[j]);
        }
        assertTrue(staged.finalizeInitialization{gas: TX_GAS}(), "finalize Balancer production facets");
        h = IBalancerHook(instance);
        assertEq(h.numTokens(), n);
        assertEq(h.pairDoorCount(), n * (n - 1) / 2);
        vm.startPrank(owner);
        IVaultFeeOracleManager(address(indexedexManager)).setVaultDexSwapFee(instance, 3e14);
        IVaultFeeOracleManager(address(indexedexManager)).setUsageFeeOfVault(instance, 0);
        vm.stopPrank();
        _balancerLiquidityAndSwaps(h, args);
    }

    function _balancerLiquidityAndSwaps(IBalancerHook h, IBalancerPkg.PkgArgs memory args) private {
        uint256 n = args.tokens.length;
        uint256[] memory amounts = new uint256[](n);
        for (uint256 i; i < n; ++i) {
            SimpleMintableERC20(args.tokens[i]).mint(detfUser, 10_000 ether);
            vm.prank(detfUser);
            IERC20(args.tokens[i]).approve(address(h), type(uint256).max);
            amounts[i] = 100 ether;
        }
        (uint256 expected,) = h.previewJoinProportional(amounts);
        vm.prank(detfUser);
        (uint256 minted,) = h.joinProportional{gas: TX_GAS}(amounts, detfUser, expected, block.timestamp + 1);
        assertEq(minted, expected, "Balancer initial LP quote");
        assertGt(minted, 0);
        _balancerFlexibleLiquidity(h, args);
        _balancerNonProportionalLiquidity(h, args);
        for (uint256 i; i < n; ++i) {
            for (uint256 j; j < n; ++j) {
                if (i != j) _balancerSwapRoute(h, args.tokens[i], args.tokens[j]);
            }
        }
        _balancerPoolManagerSwaps(h, args.tokens[0], args.tokens[n - 1]);
        _balancerSwapRoundTrip(h, args.tokens[0], args.tokens[n - 1]);
        uint256 exitShares = minted / 3;
        uint256[] memory outputs = h.previewExitProportional(exitShares);
        uint256[] memory beforeBalances = new uint256[](n);
        for (uint256 i; i < n; ++i) beforeBalances[i] = IERC20(args.tokens[i]).balanceOf(detfUser);
        vm.prank(detfUser);
        uint256[] memory actual = h.exitProportional{gas: TX_GAS}(exitShares, detfUser, outputs, block.timestamp + 1);
        for (uint256 i; i < n; ++i) {
            assertEq(actual[i], outputs[i], "Balancer exit quote");
            assertEq(IERC20(args.tokens[i]).balanceOf(detfUser) - beforeBalances[i], outputs[i], "Balancer settled exit");
            if (args.standardExchanges[i] == address(0)) continue;
            uint256 dust = IERC20(args.tokens[i]).balanceOf(address(h));
            if (dust > BalancerRepo.MAX_DUST_WEI) {
                assertEq(IStandardExchangeIn(args.standardExchanges[i]).previewExchangeIn(
                    IERC20(args.tokens[i]), dust - BalancerRepo.MAX_DUST_WEI, IERC20(args.standardExchanges[i])
                ), 0, "no bufferable excess above retained dust");
            }
            assertEq(IERC20(args.tokens[i]).allowance(address(h), args.standardExchanges[i]), 0, "pair allowance cleared");
            assertEq(IERC20(args.standardExchanges[i]).allowance(address(h), args.standardExchanges[i]), 0, "share allowance cleared");
        }
    }

    function _balancerFlexibleLiquidity(IBalancerHook h, IBalancerPkg.PkgArgs memory args) private {
        uint256 n = args.tokens.length;
        uint256[] memory amounts = new uint256[](n);
        bool[] memory flags = new bool[](n);
        for (uint256 i; i < n; ++i) {
            amounts[i] = 5 ether;
            flags[i] = args.standardExchanges[i] != address(0) && i % 2 == 0;
            if (flags[i]) {
                vm.startPrank(detfUser);
                IERC20(args.tokens[i]).approve(args.standardExchanges[i], amounts[i]);
                amounts[i] = IStandardExchangeIn(args.standardExchanges[i]).exchangeIn{gas: TX_GAS}(
                    IERC20(args.tokens[i]), amounts[i], IERC20(args.standardExchanges[i]), 1,
                    detfUser, false, block.timestamp + 1
                );
                IERC20(args.standardExchanges[i]).approve(address(h), amounts[i]);
                vm.stopPrank();
            }
        }
        IBalancerUnits units = IBalancerUnits(address(h));
        (uint256 expected, uint256[] memory used) = units.previewJoinProportionalFlexible(amounts, flags);
        uint256[] memory beforeBalances = _balancerUnitBalances(args, flags);
        vm.prank(detfUser);
        (uint256 minted, uint256[] memory actual) = units.joinProportionalFlexible{gas: TX_GAS}(
            amounts, flags, detfUser, expected, block.timestamp + 1
        );
        assertEq(minted, expected, "mixed pair/share join LP");
        assertEq(actual, used, "mixed pair/share join units");
        uint256[] memory afterBalances = _balancerUnitBalances(args, flags);
        for (uint256 i; i < n; ++i) {
            assertEq(beforeBalances[i] - afterBalances[i], used[i], "mixed pair/share input custody");
            flags[i] = args.standardExchanges[i] != address(0);
        }
        uint256[] memory output = units.previewExitProportionalFlexible(minted, flags);
        beforeBalances = _balancerUnitBalances(args, flags);
        vm.prank(detfUser);
        actual = units.exitProportionalFlexible{gas: TX_GAS}(minted, detfUser, flags, output, block.timestamp + 1);
        assertEq(actual, output, "share exit quote");
        afterBalances = _balancerUnitBalances(args, flags);
        for (uint256 i; i < n; ++i) assertEq(afterBalances[i] - beforeBalances[i], output[i], "share exit custody");
    }

    function _balancerUnitBalances(IBalancerPkg.PkgArgs memory args, bool[] memory flags)
        private view returns (uint256[] memory balances)
    {
        balances = new uint256[](flags.length);
        for (uint256 i; i < flags.length; ++i) {
            balances[i] = IERC20(flags[i] ? args.standardExchanges[i] : args.tokens[i]).balanceOf(detfUser);
        }
    }

    function _balancerNonProportionalLiquidity(IBalancerHook h, IBalancerPkg.PkgArgs memory args) private {
        uint256 n = args.tokens.length;
        uint256[] memory amounts = new uint256[](n);
        amounts[0] = 1 ether;
        amounts[n - 1] = 2 ether;
        uint256[] memory beforeBalances = _balancerUnitBalances(args, new bool[](n));
        uint256 beforeLp = IERC20(address(h)).balanceOf(detfUser);
        uint256 expected = h.previewJoinUnbalanced(amounts);
        assertGt(expected, 0, "unbalanced join quote");
        vm.prank(detfUser);

        assertEq(minted, expected, "unbalanced join LP");
        assertEq(IERC20(address(h)).balanceOf(detfUser) - beforeLp, expected, "unbalanced join LP custody");
        for (uint256 i; i < n; ++i) {
            assertEq(beforeBalances[i] - IERC20(args.tokens[i]).balanceOf(detfUser), amounts[i], "unbalanced input custody");
        }

        assertEq(IERC20(output).balanceOf(detfUser) - beforeOut, expected, "single exit custody");
        assertEq(IERC20(address(h)).balanceOf(detfUser), beforeLp, "single exit LP custody");
        _balancerExactOutputLiquidity(h, output);
    }

    function _balancerQuoteDuringUnlock(IBalancerHook h, address tokenIn, address tokenOut, uint256 amount, bool exactOut)
        private returns (uint256)
    {
        // V4 SE previews follow the current PoolManager gate, just like settlement.
        return abi.decode(pm.unlock{gas: TX_GAS}(abi.encode(h, tokenIn, tokenOut, amount, exactOut)), (uint256));
    }

    function unlockCallback(bytes calldata data) external returns (bytes memory) {
        require(msg.sender == address(pm), "PoolManager only");
        (IBalancerHook h, address tokenIn, address tokenOut, uint256 amount, bool exactOut) =
            abi.decode(data, (IBalancerHook, address, address, uint256, bool));
        return abi.encode(exactOut ? h.previewSwapExactOut(tokenIn, tokenOut, amount)
            : h.previewSwapExactIn(tokenIn, tokenOut, amount));
    }

    function _rehearse(uint256 family, uint256 backend, bool restricted) internal {
        uint256 pairCount = family == 3 ? 2 : family + 1;
        for (uint256 i; i < pairCount; ++i) {
            address pair = address(new SimpleMintableERC20("Rehearsal pair", "PAIR"));
            address other = address(new SimpleMintableERC20("Rehearsal counter asset", "OTHER"));
            pairs.push(pair);
            exchanges.push(_deploySe(backend, pair, other));
        }
        pairToken = SimpleMintableERC20(pairs[0]);
        se = exchanges[0];
        IUniswapV4Detf.PkgArgs memory args = _nLegDetfArgs(pairs.length);
        args.ownerOnlyLiquidity = restricted;
        args.name = "Rehearsal DETF";
        args.symbol = "DETF";
        if (backend == 0) {
            // V2 vault shares include a virtual-share precision offset. Set the
            // bootstrap denomination from the SE quote so stable legs start balanced.
            for (uint256 i; i < pairs.length; ++i) {
                uint256 shares = IStandardExchangeIn(exchanges[i]).previewExchangeIn(
                    IERC20(pairs[i]), 1 ether, IERC20(exchanges[i])
                );
                args.creationPairPerDetfWad[i] = 1e36 / shares;
                assertGt(args.creationPairPerDetfWad[i], 0, "nonzero bootstrap denomination");
            }
        }
        if (family == 0) detf = _deployHookThenDetf(args);
        else if (family == 1) detf = _deployWeighted(args);
        else if (family == 2) detf = _deployQuad(args);
        else detf = _deployOrbital(args);
        detfInfo = IUniswapV4Detf(detf);
        detfExchangeIn = IStandardExchangeIn(detf);
        assertEq(detfInfo.ownerOnlyLiquidity(), restricted, "DETF deployment policy");
        assertEq(IUniswapV4HookLiquidityPolicy(reserveHook).ownerOnlyLiquidity(), restricted, "actual hook policy agrees");
        assertEq(IMultiStepOwnable(reserveHook).owner(), detf, "DETF remains hook administrator");
        assertEq(IERC20Metadata(detf).symbol(), "DETF");
        assertEq(IERC20Metadata(detf).decimals(), 9);
        vm.prank(owner);
        IVaultFeeOracleManager(address(indexedexManager)).setVaultBondTerms(detf, BondTerms({
            minLockDuration: DEFAULT_MIN_LOCK, maxLockDuration: DEFAULT_MAX_LOCK,
            minBonusPercentage: 0, maxBonusPercentage: 0.5e18
        }));
        for (uint256 i; i < pairs.length; ++i) {
            SimpleMintableERC20(pairs[i]).mint(detfUser, 1_000_000 ether);
            vm.prank(detfUser);
            IERC20(pairs[i]).approve(detf, type(uint256).max);
        }
        vm.prank(detfUser);
        IERC20(detf).approve(detf, type(uint256).max);
        _lifecycle();
    }

    function _deploySe(uint256 backend, address pair, address other) internal returns (address) {
        if (backend == 0) {
            IUniswapV2Pair pool = IUniswapV2Pair(IUniswapV2Factory(V2_FACTORY).createPair(pair, other));
            SimpleMintableERC20(pair).mint(address(pool), 1_000_000 ether);
            SimpleMintableERC20(other).mint(address(pool), 1_000_000 ether);

            return IUniswapV2StandardExchangeDFPkg(_pin("phase05_stage06_uniswap_v2_standard_exchange_pkg.json", ".uniV2SePkg", true)).deployVault(pool);
        }
        if (backend == 1) {
            IUniswapV3Pool pool = SeDeploy.createUniv3PoolOneToOne(IUniswapV3Factory(V3_FACTORY), pair, other, 3000);
            SeDeploy.seedUniv3Pool(pool);
            address vault_ = SeDeploy.deployUniv3Vault(IUniswapV3StandardExchangeDFPkg(_pin("phase05_stage04_uniswap_v3_standard_exchange_pkg.json", ".uniV3SePkg", true)), pool);

        }

        assertTrue(morpho.isLltvEnabled(0), "live supply-only LLTV enabled");

        // so an unexpected oracle call fails instead of supplying a fabricated price.

    }

    function _sorted(address predicted) internal view returns (address[] memory tokens, address[] memory ses) {
        tokens = new address[](pairs.length + 1);
        ses = new address[](tokens.length);
        tokens[0] = predicted;
        for (uint256 i; i < pairs.length; ++i) tokens[i + 1] = pairs[i];
        for (uint256 i; i < tokens.length; ++i) {
            for (uint256 j = i + 1; j < tokens.length; ++j) {
                if (tokens[i] > tokens[j]) (tokens[i], tokens[j]) = (tokens[j], tokens[i]);
            }
            for (uint256 j; j < pairs.length; ++j) if (tokens[i] == pairs[j]) ses[i] = exchanges[j];
        }
    }

    function _deployWeighted(IUniswapV4Detf.PkgArgs memory args) internal returns (address) {
        address predicted = _predictDetf(args);
        (address[] memory tokens, address[] memory ses) = _sorted(predicted);
        uint256[] memory weights = new uint256[](3);
        for (uint256 i; i < 3; ++i) weights[i] = tokens[i] == predicted ? 4e17 : 3e17;
        IWeightedPkg pkg = IWeightedPkg(_pin("phase06_stage04_weighted_buffer_hook_pkg.json", ".weightedHookPkg", true));
        IWeightedPkg.PkgArgs memory h = IWeightedPkg.PkgArgs({
            poolManager: address(pm), feeOracle: address(indexedexManager), n: 3,
            tokens: tokens, weights: weights, standardExchanges: ses, rateProviders: new address[](3),
            tokenDecimals: Decimals.tokenDecimals(tokens, predicted), seDecimals: Decimals.seDecimals(ses),
            ownerOnlyLiquidity: args.ownerOnlyLiquidity, owner: predicted
        });
        reserveHook = WeightedFactory.deployHook(pkg, h, WeightedFactory.findMineNonce(hookFactory, pkg, h));
        return _finalizeAndDeploy(args, predicted, tokens);
    }

    function _deployOrbital(IUniswapV4Detf.PkgArgs memory args) private returns (address) {

    function _finalizeAndDeploy(IUniswapV4Detf.PkgArgs memory args, address predicted, address[] memory tokens) internal returns (address result) {

    }

    function _checkSyRoundTrip(IStandardizedYield sy_, uint256 amount_, bool raw_) internal {
        assertEq(sy_.decimals(), 9, "native SY precision");
        assertTrue(sy_.isValidTokenIn(detf), "DETF deposit is discoverable");
        assertTrue(sy_.isValidTokenOut(detf), "DETF redemption is discoverable");
        uint256 userBefore_ = IERC20(detf).balanceOf(detfUser);
        uint256 supplyBefore_ = IERC20(detf).totalSupply();
        uint256 ownedLpBefore_ = IERC20(reserveHook).balanceOf(detfInfo.bondNftVault());
        uint256 depositQuote_ = sy_.previewDeposit(detf, amount_);
        assertGt(depositQuote_, 0, "payable SY deposit");
        vm.startPrank(detfUser);
        IERC20(detf).approve(address(sy_), amount_);
        uint256 shares_ = sy_.deposit{gas: TX_GAS}(detfUser, detf, amount_, depositQuote_);
        assertEq(shares_, depositQuote_, "SY deposit quote equals execution");
        assertEq(sy_.balanceOf(detfUser), shares_, "actual static SY shares");
        uint256 redemptionQuote_ = sy_.previewRedeem(detf, shares_);
        assertGt(redemptionQuote_, 0, "payable SY redemption");
        uint256 redeemed_ = sy_.redeem{gas: TX_GAS}(detfUser, shares_, detf, redemptionQuote_, false);
        vm.stopPrank();
        assertEq(redeemed_, redemptionQuote_, "SY redemption quote equals execution");
        if (raw_) assertEq(redeemed_, amount_, "raw SY round trip is one to one");
        assertEq(IERC20(detf).balanceOf(detfUser), userBefore_ - amount_ + redeemed_, "actual SY recipient delta");
        assertEq(sy_.balanceOf(detfUser), 0, "all user SY redeemed");
        assertEq(sy_.totalSupply(), 0, "no residual SY liability");
        assertEq(IERC20(detf).allowance(detfUser, address(sy_)), 0, "exact SY approval consumed");
        assertEq(IERC20(detf).totalSupply(), supplyBefore_, "SY conversion creates no DETF supply");
        assertEq(IERC20(reserveHook).balanceOf(detfInfo.bondNftVault()), ownedLpBefore_, "SY conversion preserves protocol LP");
    }

        assertEq(actual, quote, "collector proportional redemption preview");
        for (uint256 i; i < tokens.length; ++i) assertEq(IERC20(tokens[i]).balanceOf(recipient), actual[i], "collector actual settlement");
        assertEq(lp.balanceOf(detfInfo.bondNftVault()), owned, "external fee LP redemption preserves protocol-owned LP");
        assertEq(lp.allowance(address(collector), reserveHook), 0, "collector clears LP approval");
    }

    function _activatePositionSe(address vault_, address pair_, address other_) internal {
        address[] memory tokens_ = new address[](2);
        tokens_[0] = pair_ < other_ ? pair_ : other_;
        tokens_[1] = pair_ < other_ ? other_ : pair_;
        uint256[] memory amounts_ = new uint256[](2);
        for (uint256 i_; i_ < 2; ++i_) {
            amounts_[i_] = 10_000 ether;
            SimpleMintableERC20(tokens_[i_]).mint(address(this), amounts_[i_]);
            IERC20(tokens_[i_]).approve(vault_, amounts_[i_]);
        }
        uint256 quote_ = IStandardExchangeInMulti(vault_).previewExchangeInManyToOne(tokens_, amounts_, IERC20(vault_));
        uint256 actual_ = IStandardExchangeInMulti(vault_).exchangeInManyToOne{gas: TX_GAS}(tokens_, amounts_, IERC20(vault_), quote_, address(this), false, block.timestamp);
        assertGt(actual_, 0, "both tokens activate position SE");
        assertEq(actual_, quote_, "position activation quote");
    }
