    function _entryPreviewMint(IERC20 tokenIn, uint256 amountIn)
        internal
        view
        returns (uint256 grossDetf, uint256 userDetf)
    {
        Repo.Storage storage s = Repo._layoutStruct();
        if (!s.isReserveLive || amountIn == 0) return (0, 0);
        if (!s.mintTable.tokens._contains(address(tokenIn))) return (0, 0);
        IStandardExchange v_ = s.mintTable.vaultOf[address(tokenIn)];
        address pair_ = _hookPairOfVault(v_);
        uint256 pairEq_;
        try IUniswapV4DetfSelfCall(address(this)).peekPairEq(address(v_), address(tokenIn), amountIn) returns (
            uint256 eq_
        ) {
            pairEq_ = eq_;
        } catch {
            return (0, 0);
        }
        if (!_mintPriceGate(pair_, true)) {
            userDetf = address(tokenIn) == address(v_)
                ? IDetfReserveQuote(s.hook).previewSwapAfterRedeem(pair_, address(this), amountIn)
                : _hook().previewSwapExactIn(pair_, address(this), pairEq_);
            return (0, userDetf);
        }
        grossDetf = _quoteMintGross(pair_, pairEq_);
        MintSplit memory split_ = _splitMintedDetf(grossDetf);
        userDetf = split_.userDetf;
    }

    function _entryPeekPairEq(address vault_, address tokenIn_, uint256 amountIn_) internal view returns (uint256) {
        return _pairEq(IStandardExchange(vault_), IERC20(tokenIn_), amountIn_);
    }

    function _entryExchangeIn(
        IERC20 tokenIn_,
        uint256 amountIn_,
        IERC20 tokenOut_,
        uint256 minAmountOut_,
        address recipient_,
        bool pretransferred_,
        uint256 deadline_
    ) internal virtual nonReentrant returns (uint256 amountOut_) {
        _requireActive(deadline_, amountIn_);
        if (recipient_ == address(0)) recipient_ = msg.sender;
        address claim_ = address(Repo._layoutStruct().rebasingClaimToken);
        if (address(tokenIn_) == address(this) && address(tokenOut_) == claim_ && claim_ != address(0)) {
            return _stakePath(amountIn_, minAmountOut_, recipient_, pretransferred_, deadline_);
        }
        if (address(tokenIn_) == claim_ && claim_ != address(0)) {
            if (pretransferred_) revert ISecurePullErrors.TransferDeltaInsufficient(amountIn_, 0);
            return _unstakePath(amountIn_, tokenOut_, minAmountOut_, recipient_, deadline_);
        }
        if (address(tokenOut_) == claim_ && claim_ != address(0)) {
            uint256 acquired_ = _mintPath(tokenIn_, amountIn_, minAmountOut_, address(this), pretransferred_, deadline_);
            return _stakeHeld(acquired_, minAmountOut_, recipient_, deadline_);
        }
        if (address(tokenIn_) == address(this)) {
            return _burnPath(amountIn_, tokenOut_, minAmountOut_, recipient_, deadline_);
        }
        if (address(tokenOut_) == address(this)) {
            return _mintPath(tokenIn_, amountIn_, minAmountOut_, recipient_, pretransferred_, deadline_);
        }
        revert Repo.InvalidRoute(address(tokenIn_), address(tokenOut_));
    }

    function _entryPreviewExchangeIn(IERC20 tokenIn_, uint256 amountIn_, IERC20 tokenOut_)
        internal
        view
        virtual
        returns (uint256 amountOut_)
    {
        address claim_ = address(Repo._layoutStruct().rebasingClaimToken);
        if (address(tokenIn_) == address(this) && address(tokenOut_) == claim_ && claim_ != address(0)) {
            return amountIn_;
        }
        if (address(tokenIn_) == claim_ && claim_ != address(0)) {
            if (address(tokenOut_) == address(this)) return amountIn_;
            return _entryPreviewBurn(amountIn_, tokenOut_);
        }
        if (address(tokenOut_) == claim_ && claim_ != address(0)) {
            (, amountOut_) = _entryPreviewMint(tokenIn_, amountIn_);
            return amountOut_;
        }
        if (address(tokenIn_) == address(this)) {
            return _entryPreviewBurn(amountIn_, tokenOut_);
        }
        if (address(tokenOut_) == address(this)) {
            (, amountOut_) = _entryPreviewMint(tokenIn_, amountIn_);
            return amountOut_;
        }
        return 0;
    }

    function _mintPath(
        IERC20 tokenIn_,
        uint256 amountIn_,
        uint256 minUserDetf_,
        address recipient_,
        bool pretransferred_,
        uint256 deadline_
    ) internal returns (uint256 userOut_) {
        _requireNotDisabled();
        _requireActive(deadline_, amountIn_);
        _requireReserveLive();
        if (recipient_ == address(0)) recipient_ = msg.sender;
        Repo.Storage storage s = Repo._layoutStruct();
        if (!s.mintTable.tokens._contains(address(tokenIn_))) {
            revert Repo.InvalidRoute(address(tokenIn_), address(this));
        }
        _realizeExpansionIfNeeded();
        if (!_isMintingAllowedToken(tokenIn_)) {
            return _swapMintPath(tokenIn_, amountIn_, minUserDetf_, recipient_, pretransferred_, deadline_);
        }
        IStandardExchange v_ = s.mintTable.vaultOf[address(tokenIn_)];
        address pair_ = _hookPairOfVault(v_);
        uint256 pulled_ = _pullToken(tokenIn_, amountIn_, pretransferred_);
        uint256 pairEq_ = _pairEq(v_, tokenIn_, pulled_);
        uint256 gross_ = _quoteMintGross(pair_, pairEq_);
        if (gross_ == 0) revert Repo.InvalidRoute(pair_, address(this));
        // Preview sizes join on the hook pair. Execute the pair when the user
        // already paid that pair so Uni V4 SE share unwrap does not transferFrom the hook.
        if (address(tokenIn_) == pair_) {
            IERC20(pair_).forceApprove(s.hook, pulled_);
            uint256 lpMint_ = _hook().joinSingleAssetExactIn(pair_, pulled_, _bondLpHolder(), 0, deadline_);
            IERC20(pair_).forceApprove(s.hook, 0);
            lpMint_;
        } else {
            uint256 shareAmt_ = _toShare(v_, tokenIn_, pulled_, deadline_);
            _joinShare(shareAmt_, address(v_));
        }
        MintSplit memory split_ = _splitMintedDetf(gross_);
        if (split_.userDetf < minUserDetf_) {
            revert IStandardExchangeErrors.MinAmountNotMet(minUserDetf_, split_.userDetf);
        }
        _mintDetf(recipient_, split_.userDetf);
        _fundStakingRewards(split_.inventoryDetf);
        _trySweepDust();
        _syncAllExpectedHoldReserves();
        return split_.userDetf;
    }

    /// @dev Explicit price-gate fallback: acquire existing pool DETF without issuance.
    function _swapMintPath(
        IERC20 tokenIn_, uint256 amountIn_, uint256 minimum_, address recipient_,
        bool pretransferred_, uint256 deadline_
    ) private returns (uint256 out_) {
        IStandardExchange vault_ = Repo._layoutStruct().mintTable.vaultOf[address(tokenIn_)];
        IERC20 pair_ = IERC20(_hookPairOfVault(vault_));
        uint256 received_ = _pullToken(tokenIn_, amountIn_, pretransferred_);
        if (address(tokenIn_) != address(pair_)) {
            received_ = _nestedExchangeInPush(
                IStandardExchangeIn(address(vault_)), tokenIn_, received_, pair_, 0, address(this), deadline_
            );
        }
        out_ = _swapOwned(pair_, IERC20(address(this)), received_, minimum_, deadline_);
        IERC20(address(this)).safeTransfer(recipient_, out_);
        _syncAllExpectedHoldReserves();
    }

    /// @dev Measure only this swap's acquired output; idle protocol assets are never a payout subsidy.
    function _swapOwned(IERC20 tokenIn_, IERC20 tokenOut_, uint256 amount_, uint256 minimum_, uint256 deadline_)
        private returns (uint256 out_)
    {
        address hook_ = Repo._layoutStruct().hook;
        uint256 before_ = tokenOut_.balanceOf(address(this));
        tokenIn_.forceApprove(hook_, amount_);
        _hook().ownerSwapExactIn(address(tokenIn_), address(tokenOut_), amount_, minimum_, deadline_);
        tokenIn_.forceApprove(hook_, 0);
        out_ = tokenOut_.balanceOf(address(this)) - before_;
        if (out_ < minimum_) revert IStandardExchangeErrors.MinAmountNotMet(minimum_, out_);
    }

    /// @dev Live-only, ungated vs mint/burn thresholds (D22). Joins DETF self-leg,
    ///      credits id 0, then prices claim issuance against its pre-credit principal.
    /// @dev Direct staking depends only on funded DETF, without any reserve gate.
    function _stakePath(uint256 amount_, uint256 minimum_, address recipient_, bool pretransferred_, uint256 deadline_)
        internal returns (uint256)
    {
        _requireActive(deadline_, amount_);
        _realizeExpansionIfNeeded();
        uint256 received_ = _pullToken(IERC20(address(this)), amount_, pretransferred_);
        return _stakeHeld(received_, minimum_, recipient_, deadline_);
    }

    function _stakeHeld(uint256 amount_, uint256 minimum_, address recipient_, uint256 deadline_)
        internal returns (uint256 out_)
    {
        address staking_ = address(Repo._layoutStruct().rebasingClaimToken);
        IERC20(address(this)).forceApprove(staking_, amount_);
        out_ = IStakedDETF(staking_).exchangeIn(
            IERC20(address(this)), amount_, IERC20(staking_), minimum_, recipient_, false, deadline_
        );
        IERC20(address(this)).forceApprove(staking_, 0);
        _syncAllExpectedHoldReserves();
    }

    function _unstakePath(uint256 amount_, IERC20 tokenOut_, uint256 minimum_, address recipient_, uint256 deadline_)
        internal returns (uint256 out_)
    {
        _realizeExpansionIfNeeded();
        address staking_ = address(Repo._layoutStruct().rebasingClaimToken);
        IERC20(staking_).safeTransferFrom(msg.sender, address(this), amount_);
        bool direct_ = address(tokenOut_) == address(this);
        out_ = IStakedDETF(staking_).exchangeIn(
            IERC20(staking_), amount_, IERC20(address(this)), direct_ ? minimum_ : 0,
            direct_ ? recipient_ : address(this), false, deadline_
        );
        if (!direct_) out_ = _burnHeld(out_, tokenOut_, minimum_, recipient_, deadline_);
        _syncAllExpectedHoldReserves();
    }

    function _entryPreviewBurn(uint256 detfIn, IERC20 tokenOut) internal view returns (uint256 amountOut) {
        Repo.Storage storage s = Repo._layoutStruct();
        if (!s.isReserveLive || detfIn == 0) return 0;
        if (!s.burnTable.tokens._contains(address(tokenOut))) return 0;
        IStandardExchange v_ = s.burnTable.vaultOf[address(tokenOut)];
        address pair_ = _hookPairOfVault(v_);
        if (!_burnPriceGate(pair_, true)) {
            uint256 acquired_ = _hook().previewSwapExactIn(address(this), pair_, detfIn);
            if (address(tokenOut) == pair_) return acquired_;
            return v_.previewExchangeIn(IERC20(pair_), acquired_, tokenOut);
        }
        uint256 supply_ = ERC20Repo._totalSupply() + _pendingExpansionDetf();
        if (supply_ == 0) return 0;
        uint256 lpOut_ = Math.mulDiv(detfIn, _protocolLp(), supply_);
        if (lpOut_ == 0) return 0;
        // Match _entryBurn(): exitProportional + rejoin DETF + pay remaining pair (H10: not exitSingleAsset).
        uint256 residual_ = _previewPropPairResidual(lpOut_, pair_);
        if (address(tokenOut) == pair_) return residual_;
        if (address(tokenOut) == address(v_)) {
            try v_.previewExchangeIn(IERC20(pair_), residual_, IERC20(address(v_))) returns (uint256 sh_) {
                return sh_;
            } catch {
                return 0;
            }
        }
        try v_.previewExchangeIn(IERC20(pair_), residual_, tokenOut) returns (uint256 o_) {
            return o_;
        } catch {
            return 0;
        }
    }

    function _previewPropPairResidual(uint256 lpOut_, address pair_) private view returns (uint256 residual_) {
        uint256[] memory amounts_ = _hook().previewExitProportional(lpOut_);
        address[] memory tokens_ = _hook().tokens();
        uint256 n_ = amounts_.length < tokens_.length ? amounts_.length : tokens_.length;
        for (uint256 i; i < n_; ++i) {
            if (tokens_[i] == pair_) return amounts_[i];
        }
    }

    function _burnPath(uint256 detfIn_, IERC20 tokenOut_, uint256 minAmountOut_, address recipient_, uint256 deadline_)
        internal
        returns (uint256 amountOut_)
    {
        _requireActive(deadline_, detfIn_);
        _realizeExpansionIfNeeded();
        IERC20(address(this)).safeTransferFrom(msg.sender, address(this), detfIn_);
        return _burnHeld(detfIn_, tokenOut_, minAmountOut_, recipient_, deadline_);
    }

    function _burnHeld(uint256 detfIn_, IERC20 tokenOut_, uint256 minAmountOut_, address recipient_, uint256 deadline_)
        internal returns (uint256 amountOut_)
    {
        _requireActive(deadline_, detfIn_);
        _requireReserveLive();
        if (recipient_ == address(0)) recipient_ = msg.sender;
        Repo.Storage storage s = Repo._layoutStruct();
        if (!s.burnTable.tokens._contains(address(tokenOut_))) {
            revert Repo.InvalidRoute(address(this), address(tokenOut_));
        }
        _realizeExpansionIfNeeded();
        if (!_isBurningAllowedToken(tokenOut_)) {
            return _swapBurnPath(detfIn_, tokenOut_, minAmountOut_, recipient_, deadline_);
        }
        uint256 supply_ = ERC20Repo._totalSupply();
        uint256 lpOut_ = Math.mulDiv(detfIn_, _protocolLp(), supply_);
        if (lpOut_ == 0) revert Repo.ZeroAmount();
        _burnDetf(address(this), detfIn_);
        uint256[] memory exitAmts_ = _exitBurnLp(lpOut_, deadline_);
        amountOut_ = _payBurnOut(tokenOut_, recipient_, deadline_, exitAmts_);
        if (amountOut_ < minAmountOut_) {
            revert IStandardExchangeErrors.MinAmountNotMet(minAmountOut_, amountOut_);
        }
        _rejoinExitDetf(exitAmts_, deadline_);
        _returnLeftoverLp();
        _trySweepDust();
        _syncAllExpectedHoldReserves();
    }

    /// @dev Explicit burn-gate fallback transfers existing DETF into the reserve without burning it.
    function _swapBurnPath(uint256 detfIn_, IERC20 tokenOut_, uint256 minimum_, address recipient_, uint256 deadline_)
        private returns (uint256 out_)
    {
        IStandardExchange vault_ = Repo._layoutStruct().burnTable.vaultOf[address(tokenOut_)];
        IERC20 pair_ = IERC20(_hookPairOfVault(vault_));
        uint256 acquired_ = _swapOwned(IERC20(address(this)), pair_, detfIn_, 0, deadline_);
        if (address(tokenOut_) == address(pair_)) {
            out_ = acquired_;
            pair_.safeTransfer(recipient_, out_);
        } else {
            out_ = _nestedExchangeInPush(
                IStandardExchangeIn(address(vault_)), pair_, acquired_, tokenOut_, minimum_, recipient_, deadline_
            );
        }
        if (out_ < minimum_) revert IStandardExchangeErrors.MinAmountNotMet(minimum_, out_);
        _syncAllExpectedHoldReserves();
    }

    function _exitBurnLp(uint256 lpOut_, uint256 deadline_) private returns (uint256[] memory amounts_) {
        Repo.Storage storage s = Repo._layoutStruct();
        _pullNftLp(lpOut_);
        IERC20(s.hook).forceApprove(s.hook, lpOut_);
        amounts_ = _hook().exitProportional(lpOut_, address(this), new uint256[](_hook().tokens().length), deadline_);
        IERC20(s.hook).forceApprove(s.hook, 0);
    }

    function _rejoinExitDetf(uint256[] memory amounts_, uint256 deadline_) private {
        address[] memory tokens_ = _hook().tokens();
        uint256 n_ = amounts_.length < tokens_.length ? amounts_.length : tokens_.length;
        for (uint256 i; i < n_; ++i) {
            if (tokens_[i] == address(this) && amounts_[i] > 0) {
                uint256 have_ = IERC20(address(this)).balanceOf(address(this));
                uint256 joinAmt_ = amounts_[i] < have_ ? amounts_[i] : have_;
                if (joinAmt_ == 0) continue;
                IERC20(address(this)).forceApprove(Repo._layoutStruct().hook, joinAmt_);
                _hook().joinSingleAssetExactIn(address(this), joinAmt_, _bondLpHolder(), 0, deadline_);
                IERC20(address(this)).forceApprove(Repo._layoutStruct().hook, 0);
            }
        }
    }

    function _payBurnOut(IERC20 tokenOut_, address recipient_, uint256 deadline_, uint256[] memory withdrawn_)
        private returns (uint256 amountOut_)
    {
        Repo.Storage storage s = Repo._layoutStruct();
        IStandardExchange v_ = s.burnTable.vaultOf[address(tokenOut_)];
        address pair_ = _hookPairOfVault(v_);
        address[] memory tokens_ = _hook().tokens();
        uint256 residual_;
        for (uint256 i; i < tokens_.length; ++i) {
            if (tokens_[i] == pair_) residual_ = withdrawn_[i];
        }
        if (address(tokenOut_) == pair_) {
            amountOut_ = residual_;
            if (amountOut_ > 0) IERC20(pair_).safeTransfer(recipient_, amountOut_);
            return amountOut_;
        }
        return _nestedExchangeInPush(
            IStandardExchangeIn(address(v_)), IERC20(pair_), residual_, tokenOut_, 0, recipient_, deadline_
        );
    }

    function _entryBond(
        IERC20 tokenIn,
        uint256 amountIn,
        uint256 lockDuration,
        address recipient,
        bool pretransferred,
        uint256 deadline
    ) internal nonReentrant returns (uint256 tokenId, uint256 shares) {
        _requireNotDisabled();
        _requireActive(deadline, amountIn);
        if (recipient == address(0)) recipient = msg.sender;
        Repo.Storage storage s = Repo._layoutStruct();
        if (s.isReserveLive) {
            _realizeExpansionIfNeeded();
            if (address(tokenIn) == s.hook) {
                // A pretransfer is already protocol inventory and cannot prove this caller's ownership.
                if (pretransferred) revert Repo.NotAuthorized(msg.sender);
                return _purchaseReserveLpBond(amountIn, lockDuration, recipient);
            }
            if (!s.bondTable.tokens._contains(address(tokenIn))) {
                revert Repo.InvalidRoute(address(tokenIn), address(this));
            }
        } else {
            _rejectPretransferredFirstBond(pretransferred, amountIn);
            _requireReserveWired();
        }
        BondSeat memory seat_ = _seatBond(tokenIn, amountIn, pretransferred, deadline, s.isReserveLive);
        (tokenId, shares) = _purchaseBond(seat_, lockDuration, recipient);
        _trySweepDust();
        _syncAllExpectedHoldReserves();
    }

    function _purchaseReserveLpBond(uint256 amount_, uint256 lockDuration_, address recipient_)
        private returns (uint256 tokenId_, uint256 shares_)
    {
        uint256 duration_ = _effectiveLockDuration(lockDuration_);
        // Expansion has settled. Snapshot valuation before the caller's LP enters protocol custody.
        MintSplit memory split_ = _splitBondDetf(_quoteReserveLpBond(amount_, duration_), 0);
        if (split_.userDetf == 0) revert Repo.ZeroAmount();
        IERC20 lp_ = IERC20(Repo._layoutStruct().hook);
        shares_ = _pullToken(lp_, amount_, false);
        if (shares_ != amount_) revert ISecurePullErrors.TransferDeltaInsufficient(amount_, shares_);
        // The complete acquired LP belongs to the protocol, including its existing DETF inventory.
        lp_.safeTransfer(_bondLpHolder(), shares_);
        tokenId_ = _openBondNft(split_.userDetf, duration_, recipient_);
        _fundStakingRewards(split_.inventoryDetf);
        _syncAllExpectedHoldReserves();
    }

    function _purchaseBond(BondSeat memory seat_, uint256 lockDuration, address recipient)
        private returns (uint256 tokenId, uint256 shares)
    {
        Repo.Storage storage s = Repo._layoutStruct();
        bool first_ = !s.isReserveLive;
        uint256 duration_ = _effectiveLockDuration(lockDuration);
        uint256 pairEq_ = _bondPairEq(seat_);
        uint256 g_ = _quoteBondG(seat_.pair, pairEq_);
        if (g_ == 0) revert Repo.FirstBondBelowMinimumLiquidity();
        MintSplit memory split_ = _splitBondDetf(_quoteBondPurchase(seat_.pair, pairEq_, duration_), g_);
        shares = _joinBond(g_, seat_);
        if (first_ && shares == 0) revert Repo.FirstBondBelowMinimumLiquidity();
        if (first_) {
            Repo._setReserveLive();
            s.epochAnchor = block.timestamp;
            s.lastExpansionTimestamp = block.timestamp;
        }
        tokenId = _openBondNft(split_.userDetf, duration_, recipient);
        if (first_) emit IUniswapV4Detf.ReserveLive(tokenId, shares);
        _fundStakingRewards(split_.inventoryDetf);
    }

    function _seatBond(IERC20 tokenIn_, uint256 amountIn_, bool pretransferred_, uint256 deadline_, bool live_)
        private
        returns (BondSeat memory seat_)
    {
        uint256 pulled_ = _pullToken(tokenIn_, amountIn_, pretransferred_);
        Repo.Storage storage s = Repo._layoutStruct();
        if (live_) {
            seat_.vault = s.bondTable.vaultOf[address(tokenIn_)];
            seat_.pair = _hookPairOfVault(seat_.vault);
        } else {
            seat_.pair = _firstBondPair(address(tokenIn_));
            seat_.vault = IStandardExchange(IUniswapV4SeBufferHook(s.hook).standardExchangeOf(seat_.pair));
        }
        if (address(tokenIn_) == seat_.pair || address(tokenIn_) == address(seat_.vault)) {
            seat_.capital = address(tokenIn_);
            seat_.amount = pulled_;
        } else {
            seat_.capital = address(seat_.vault);
            seat_.amount = _toShare(seat_.vault, tokenIn_, pulled_, deadline_);
        }
    }

    function _bondPairEq(BondSeat memory seat_) private view returns (uint256) {
        if (seat_.capital == seat_.pair) return seat_.amount;
        return _pairEq(seat_.vault, IERC20(seat_.capital), seat_.amount);
    }

    function _joinBond(uint256 g_, BondSeat memory seat_) private returns (uint256 shares_) {
        _mintDetf(address(this), g_);
        Repo.Storage storage s = Repo._layoutStruct();
        if (!s.isReserveLive && _hook().firstJoinMustBeFullBook()) {
            return _joinFirstBondFullBook(g_, seat_);
        }
        address[] memory joinToks_ = new address[](2);
        uint256[] memory joinAmts_ = new uint256[](2);
        joinToks_[0] = address(this);
        joinToks_[1] = seat_.capital;
        joinAmts_[0] = g_;
        joinAmts_[1] = seat_.amount;
        shares_ = _joinUnbalanced(joinToks_, joinAmts_);
    }

    /// @notice First bond: joinUnbalanced every requiredFirstBondTokens leg (pair or that pair's share).
    /// @dev Lead capital is already pulled. Other non-DETF legs are pulled from msg.sender at opening-rate size.
    function _joinFirstBondFullBook(uint256 g_, BondSeat memory seat_) private returns (uint256 shares_) {
        address[] memory required_ = _hook().requiredFirstBondTokens();
        uint256 n_ = required_.length;
        address[] memory joinToks_ = new address[](n_);
        uint256[] memory joinAmts_ = new uint256[](n_);
        Repo.Storage storage s = Repo._layoutStruct();
        for (uint256 i; i < n_; ++i) {
            address t_ = required_[i];
            if (t_ == address(this)) {
                joinToks_[i] = t_;
                joinAmts_[i] = g_;
                continue;
            }
            if (t_ == seat_.pair) {
                joinToks_[i] = seat_.capital;
                joinAmts_[i] = seat_.amount;
                continue;
            }
            uint256 need_ = _firstBondOtherPayment(g_, t_);
            joinToks_[i] = t_;
            joinAmts_[i] = _pullToken(IERC20(t_), need_, false);
        }
        shares_ = _joinUnbalanced(joinToks_, joinAmts_);
    }

    function _firstBondPair(address tokenIn_) internal view returns (address pair_) {
        Repo.Storage storage s = Repo._layoutStruct();
        if (s.hookPairTokens._contains(tokenIn_)) return tokenIn_;
        address[] storage pairs_ = s.hookPairTokens._values();
        for (uint256 i; i < pairs_.length; ++i) {
            address se_ = IUniswapV4SeBufferHook(s.hook).standardExchangeOf(pairs_[i]);
            if (tokenIn_ == se_) return pairs_[i];
        }
        if (pairs_.length == 1) return pairs_[0];
        revert Repo.InvalidRoute(tokenIn_, address(this));
    }

    function _openBondNft(uint256 principal_, uint256 lockDuration_, address recipient_)
        private
        returns (uint256 tokenId_)
    {
        Repo.Storage storage s = Repo._layoutStruct();
        if (principal_ == 0) revert Repo.ZeroAmount();
        if (address(s.bondNftVault) == address(0)) revert Repo.ReserveNotWired();
        _mintDetf(address(this), principal_);
        IERC20(address(this)).forceApprove(address(s.bondNftVault), principal_);
        tokenId_ = IDetfBondNFT(address(s.bondNftVault)).createFundedPosition(principal_, lockDuration_, recipient_);
        IERC20(address(this)).forceApprove(address(s.bondNftVault), 0);
    }

    function _entryDonate(IERC20 token, uint256 amount, bool pretransferred) internal {
        Repo.Storage storage s = Repo._layoutStruct();
        IDetfNftReserveDonation(address(s.bondNftVault))
            .donate(msg.sender, token, amount, 0, pretransferred, block.timestamp + 1);
    }

    function _entryJoinDonatedCapital(IERC20 token, uint256 amount, uint256 deadline)
        internal
        nonReentrant
        returns (uint256 lpOut)
    {
        _requireBondNft();
        _requireReserveLive();
        _requireActive(deadline, amount);
        Repo.Storage storage s = Repo._layoutStruct();
        if (address(token) == address(this)) {
            IERC20(address(this)).safeTransferFrom(msg.sender, address(this), amount);
            lpOut = _joinShare(amount, address(this));
            _trySweepDust();
            _syncAllExpectedHoldReserves();
            return lpOut;
        }
        if (address(token) == s.hook) {
            return 0;
        }
        bool allowed_ = s.donateTable.tokens._contains(address(token));
        if (!allowed_) revert Repo.InvalidRoute(address(token), address(this));
        IERC20(token).safeTransferFrom(msg.sender, address(this), amount);
        IStandardExchange v_ = s.donateTable.vaultOf[address(token)];
        // Use the same native pair join as ordinary issuance. Wrapping the pair
        // first changes the shared SE book before the hook's own intake quote.
        if (address(token) == _hookPairOfVault(v_)) {
            token.forceApprove(s.hook, amount);
            lpOut = _hook().joinSingleAssetExactIn(address(token), amount, _bondLpHolder(), 0, deadline);
            token.forceApprove(s.hook, 0);
            _trySweepDust();
            _syncAllExpectedHoldReserves();
            return lpOut;
        }
        uint256 shareAmt_ = _toShare(v_, token, amount, deadline);
        lpOut = _joinShare(shareAmt_, address(v_));
        _trySweepDust();
        _syncAllExpectedHoldReserves();
    }

    function _entryPreviewJoinDonatedCapital(IERC20 token, uint256 amount) internal view returns (uint256 lpOut) {
        Repo.Storage storage s = Repo._layoutStruct();
        if (!s.isReserveLive || amount == 0) return 0;
        if (address(token) == address(this)) {
            return _hook().previewJoinSingleAssetExactIn(address(this), amount);
        }
        if (address(token) == s.hook) return amount;
        if (!s.donateTable.tokens._contains(address(token))) return 0;
        IStandardExchange v_ = s.donateTable.vaultOf[address(token)];
        if (address(token) == _hookPairOfVault(v_)) {
            return _hook().previewJoinSingleAssetExactIn(address(token), amount);
        }
        uint256 shareAmt_ = address(token) == address(v_) ? amount : 0;
        if (shareAmt_ == 0) {
            try v_.previewExchangeIn(token, amount, IERC20(address(v_))) returns (uint256 sh_) {
                shareAmt_ = sh_;
            } catch {
                return 0;
            }
        }
        return _hook().previewJoinSingleAssetExactIn(address(v_), shareAmt_);
    }

    function _entryNotifyReserveDonated() internal {
        _requireBondNft();
        _trySweepDust();
        _syncAllExpectedHoldReserves();
    }

    function _entrySweepDust() internal nonReentrant {
        _sweepDustBody();
        _syncAllExpectedHoldReserves();
    }

    function _entryCompleteReserveBondNft() internal returns (address bondNftVault_) {
        Repo.Storage storage s = Repo._layoutStruct();
        if (s.hook == address(0)) revert Repo.ReserveNotWired();
        if (address(s.bondNftVault) != address(0)) revert Repo.ReserveBondNftAlreadyWired();
        address detf_ = address(this);
        IDETFNFTVault bondVault_ = IDETFNFTVault(
            IUniswapV4DetfBondNFTVaultDFPkg(s.bondNftVaultPkg)
                .deployVault(
                    DETFChildTokenMetadata.resolveBondName(s.bondName, ERC20Repo._name()),
                    DETFChildTokenMetadata.resolveBondSymbol(s.bondSymbol, ERC20Repo._symbol()),
                    IDetf(detf_),
                    IERC20(s.hook)
                )
        );
        uint256 detfNftId_;
        address feeTo_ = address(s.feeOracle.feeTo());
        address creator_ = s.creator;
        detfNftId_ = bondVault_.initializeReservedBondNfts(feeTo_, creator_);
        Repo._setBondNft(bondVault_);
        emit IUniswapV4Detf.ReserveBondNftWired(s.hook, address(bondVault_), detfNftId_, DETF_FEE_TO_BOND_NFT_ID);
        return address(bondVault_);
    }

    function _entryCompleteReserveClaim() internal returns (address rebasingClaimToken_) {
        Repo.Storage storage s = Repo._layoutStruct();
        if (address(s.bondNftVault) == address(0)) revert Repo.ReserveNotWired();
        if (address(s.rebasingClaimToken) != address(0)) revert Repo.ReserveClaimAlreadyWired();
        IRebasingClaimToken claimToken_ = IRebasingClaimToken(
            IRebasingClaimTokenDFPkg(s.rebasingClaimTokenPkg)
                .deployToken(
                    IDetf(address(this)),
                    s.bondNftVault,
                    s.feeOracle,
                    DETFChildTokenMetadata.resolveClaimName(s.claimName, ERC20Repo._name()),
                    DETFChildTokenMetadata.resolveClaimSymbol(s.claimSymbol, ERC20Repo._symbol())
                )
        );
        Repo._setClaim(claimToken_);
        emit IUniswapV4Detf.ReserveClaimWired(s.hook, address(claimToken_));
        return address(claimToken_);
    }

    function _firstBondOtherPayment(uint256 detf_, address pair_) internal view returns (uint256 amount_) {
        Repo.Storage storage s_ = Repo._layoutStruct();
        uint256 opening_ = s_.openingOfPair[pair_];
        if (opening_ == 0) opening_ = s_.creationOfPair[pair_];
        amount_ = DETFDecimalScaleLib.wadToNative(pair_, Math.mulDiv(detf_, opening_, 1e9));
        if (amount_ == 0) revert Repo.FirstBondBelowMinimumLiquidity();
    }

    function _entryPreviewFirstBondPayments(IERC20 tokenIn_, uint256 amount_)
        internal view returns (address[] memory tokens_, uint256[] memory amounts_)
    {
        _requireReserveWired();
        Repo.Storage storage s_ = Repo._layoutStruct();
        if (s_.isReserveLive) revert Repo.AlreadyLive();
        if (amount_ == 0) revert Repo.ZeroAmount();
        address pair_ = address(tokenIn_);
        if (!s_.hookPairTokens._contains(pair_)) revert Repo.InvalidRoute(pair_, address(this));
        if (!_hook().firstJoinMustBeFullBook()) {
            tokens_ = new address[](1); amounts_ = new uint256[](1);
            tokens_[0] = pair_; amounts_[0] = amount_; return (tokens_, amounts_);
        }
        uint256 detf_ = _quoteBondG(pair_, amount_);
        if (detf_ == 0) revert Repo.FirstBondBelowMinimumLiquidity();
        tokens_ = _entryAcceptedBondTokens(); amounts_ = new uint256[](tokens_.length);
        for (uint256 i_; i_ < tokens_.length; ++i_) {
            amounts_[i_] = tokens_[i_] == pair_ ? amount_ : _firstBondOtherPayment(detf_, tokens_[i_]);
        }
    }
